 package gr.softeng.team19.domain;
 
 import java.util.ArrayList;
 
 public class Library {
 
     private static ArrayList<ApplicationUser> users = new ArrayList<>();
 
     public static ArrayList<ApplicationUser> getUsers() {
         return users;
     }
 
     public static void addUser(ApplicationUser user){
         users.add(user);
     }
 
     public static ApplicationUser loginUser(String username, String password) {
         for (ApplicationUser user : users) {
             if (user.getUserName().equals(username) && user.getPassword().equals(password)) {
                 if (user instanceof TaxiDriver) {
                     TaxiDriver dr = (TaxiDriver) user;
                     return dr;
                 } else if (user instanceof Customer) {
                     Customer cs = (Customer) user;
                     return cs;
                 }   
                 
             }
         }
         return null;
     }
 
     private static ArrayList<TaxiBooking> bookings = new ArrayList<>();
 
     public static ArrayList<TaxiBooking> getBookings() {
         return bookings;
     }
 
     public static void addBooking(TaxiBooking booking){
         bookings.add(booking);
     }
 
     public static void updateBooking(TaxiBooking booking) {
         String bookingID = booking.getBookingID();
         int indexToReplace = -1;
         for (int i = 0; i < bookings.size(); i++) {
             if (bookings.get(i).getBookingID().equals(bookingID) ) {
                 indexToReplace = i;
                 break;
             }
         }
         if (indexToReplace != -1) {
             bookings.set(indexToReplace, booking);
 
     }
    }
 
 }

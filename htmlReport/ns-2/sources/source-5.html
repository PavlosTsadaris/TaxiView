 package gr.softeng.team19.domain;
 import java.util.ArrayList;
 
 public class DriverRating {
     private ArrayList<Integer> serviceRatings;
     private Double averageRating;
     private String customerComment;
     private String ratingID;
 
     
     public DriverRating() {
         this.serviceRatings = new ArrayList<>();
         this.averageRating = 0.0;
         this.customerComment = "";
         this.ratingID = GlobalID.getNextRatingID();
     }
     public String getRatingID() {
         return ratingID;
     }
     public void setRatingID(String ratingID) {
         this.ratingID = ratingID;
     }
 
     public ArrayList<Integer> getServiceRatings() {
         return serviceRatings;
     }
 
     public Boolean addServiceRating(Integer rating) {
         if (rating != null && rating >= 0 && rating <= 5) {
             this.serviceRatings.add(rating);
             updateAverageRating();
             return true;
         }
         return false;
     }
 
     public Double getAverageRating() {
         return averageRating;
     }
 
     private void updateAverageRating() {
         if (serviceRatings.isEmpty()) {
             this.averageRating = 0.0;
             return;
         }
         Double sum = 0.0;
         for (Integer rating : serviceRatings) {
             sum += rating;
         }
         this.averageRating = sum / serviceRatings.size();
     }
 
     public String getCustomerComment() {
         return customerComment;
     }
 
     public void setCustomerComment(String customerComment) {
         this.customerComment = customerComment;
     }
 
     public void fillDriverRating(ArrayList<Integer> costumerRatings, String comment) {
         for (Integer rating : costumerRatings) {
 
             addServiceRating(rating);
         }
         setCustomerComment(comment);
         updateAverageRating();
 
     }
 }

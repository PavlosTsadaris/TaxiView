 package gr.softeng.team19.view.history;
 
 import java.util.ArrayList;
 import java.util.List;
 
 import gr.softeng.team19.dao.TaxiBookingDAO;
 import gr.softeng.team19.domain.TaxiBooking;
 import gr.softeng.team19.memorydao.TaxiBookingDAOMemory;
 
 /**
  * Presenter for the Ride History Screen.
  * <p>
  * This class handles the business logic for fetching, filtering, and displaying
  * the ride history for a specific user (either a Passenger or a Driver).
  * It calculates the total amount spent (or earned) and manages UI updates.
  * </p>
  */
 public class HistoryPresenter {
 
     private HistoryView view;
     private TaxiBookingDAO bookingDAO;
     private String currentUsername;
 
     /**
      * Constructor for the HistoryPresenter.
      * Initializes the DAO and immediately triggers the history loading process.
      *
      * @param view     The interface for updating the UI.
      * @param username The username of the currently logged-in user (Driver or Customer).
      */
     public HistoryPresenter(HistoryView view, String username) {
         this.view = view;
         this.currentUsername = username;
         this.bookingDAO = new TaxiBookingDAOMemory();
 
         // Load history immediately upon creation
         loadHistory();
     }
 
     /**
      * Fetches all bookings from the database and filters them to find
      * those associated with the current user.
      * <p>
      * Logic:
      * 1. Retrieves all bookings.
      * 2. Checks if the current user is either the Customer or the Driver of the booking.
      * 3. Accumulates the total payment amount.
      * 4. Updates the view with the list or an empty state.
      * </p>
      */
     private void loadHistory() {
         List<TaxiBooking> allBookings = bookingDAO.findAll();
         List<TaxiBooking> userHistory = new ArrayList<>();
         double totalAmount = 0.0;
 
         if (currentUsername == null) {
             view.showEmptyState();
             return;
         }
 
         for (TaxiBooking booking : allBookings) {
             // Check if the current user is the customer of this booking
             boolean isMyCustomerRide = booking.getCustomer() != null &&
                     booking.getCustomer().getUserName().equals(currentUsername);
 
             // Check if the current user is the driver of this booking
             boolean isMyDriverRide = booking.getTaxiDriver() != null &&
                     booking.getTaxiDriver().getUserName().equals(currentUsername);
 
             if (isMyCustomerRide || isMyDriverRide) {
                 userHistory.add(booking);
 
                 // Add payment amount if available
                 if (booking.getPayment() != null) {
                     totalAmount += booking.getPayment().getAmount();
                 }
             }
         }
 
         if (userHistory.isEmpty()) {
             view.showEmptyState();
             view.showTotalAmount(0.0);
         } else {
             view.showHistoryList(userHistory);
             view.showTotalAmount(totalAmount);
         }
     }
 
     /**
      * Handles the user's request to navigate back.
      */
     public void goBack() {
         view.navigateBack();
     }
 }

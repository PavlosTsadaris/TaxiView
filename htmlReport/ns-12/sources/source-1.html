 package gr.softeng.team19.view.start.login;
 
 import gr.softeng.team19.domain.TaxiDriver;
 import gr.softeng.team19.domain.User;
 import gr.softeng.team19.memorydao.UserDAOMemory;
 
 /**
  * LoginPresenter handles the authentication logic for the login screen.
  * It validates user credentials against the UserDAOMemory and directs
  * the user to the appropriate menu based on their account type.
  */
 public class LoginPresenter {
     private LoginView view;
 
     /**
      * Connects the Presenter with the View.
      * @param view An implementation of LoginView to handle UI interactions.
      */
     public LoginPresenter(LoginView view) {
         this.view = view;
     }
 
     /**
      * Processes the login attempt.
      * Validates that fields are not empty, checks for user existence,
      * verifies the password, and handles role-based navigation.
      */
     public void onLogin() {
         String username = view.getUsername().trim();
         String password = view.getPassword().trim();
 
         if (username.isEmpty() || password.isEmpty()) {
             view.showErrorMessage("Error", "Please fill all fields.");
             return;
         }
 
         User user = new UserDAOMemory().find(username);
 
         if (user == null) {
             view.showErrorMessage("Login Failed", "User not found.");
             return;
         }
 
         if (!user.getPassword().equals(password)) {
             view.showErrorMessage("Login Failed", "Wrong Password.");
         } else {
             view.showSuccessMessage("Welcome back, " + user.getUserName() + "!");
 
             if(user instanceof TaxiDriver){
                 view.navigateToDriverMenu();
             } else {
                 view.navigateToCustomerMenu();
             }
         }
     }
 
     /**
      * Handles the intent to navigate to the registration screen.
      */
     public void onSignUp() {
         view.navigateToSignUp();
     }
 }

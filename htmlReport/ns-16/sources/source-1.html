 package gr.softeng.team19.view.viewmyprofile;
 
 import gr.softeng.team19.domain.TaxiDriver;
 import gr.softeng.team19.memorydao.TaxiDriverDAOMemory;
 import gr.softeng.team19.memorydao.UserDAOMemory;
 import gr.softeng.team19.domain.ApplicationUser;
 
 /**
  * MyProfilePresenter manages the logic for displaying user profile information.
  * It handles fetching user data from the database, formatting data for display 
  * (such as masking credit card numbers), and managing state changes during logout.
  */
 public class MyProfilePresenter {
     private MyProfileView view;
     private UserDAOMemory userDAO;
     private ApplicationUser user;
 
     /**
      * Constructs the presenter with a view and a DAO.
      * @param view The view implementation for displaying profile data.
      * @param userDAO The DAO used to fetch user information.
      */
     public MyProfilePresenter(MyProfileView view, UserDAOMemory userDAO) {
         this.view = view;
         this.userDAO = userDAO;
     }
 
     /**
      * Loads the profile data for a specific username.
      * Validates the username and fetches the user from the database.
      * @param username The unique identifier of the user to load.
      */
     public void onLoadProfile(String username) {
         if (username == null || username.isEmpty()) {
             view.showError("User not found");
             return;
         }
 
         user = userDAO.find(username);
 
         if (user != null) {
             String fullname = user.getName() + " " + user.getSurname();
             String maskedCard = maskCardNumber(user.getCreditCardNumber());
 
             view.showUserDetails(
                     fullname,
                     user.getUserName(),
                     user.getEmail(),
                     user.getPhoneNumber(),
                     user.getAddress().toString(),
                     maskedCard
             );
         } else {
             view.showError("Error: User data missing in Database.");
         }
     }
 
     /**
      * Masks the credit card number to show only the last four digits.
      * @param fullCard The raw credit card string.
      * @return A masked string or stars if the input is invalid.
      */
     private String maskCardNumber(String fullCard) {
         if (fullCard == null || fullCard.length() < 4) {
             return "****";
         }
         String lastFour = fullCard.substring(fullCard.length() - 4);
         return "**** **** **** " + lastFour;
     }
 
     /**
      * Handles the request to navigate back to the previous screen.
      */
     public void onBackRequested() {
         view.navigateBack();
     }
 
     /**
      * Handles the logout request.
      * If the current user is a TaxiDriver, their availability is set to false 
      * in the database before logging out.
      */
     public void onLogoutRequested() {
         if(user instanceof TaxiDriver){
             TaxiDriverDAOMemory taxiDriverDAO = new TaxiDriverDAOMemory();
             taxiDriverDAO.find(user.getUserName()).setAvailability(false);
         }
         view.logout();
     }
 }

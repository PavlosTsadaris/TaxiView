 package gr.softeng.team19.view.start.welcome;
 
 /**
  * WelcomePresenter handles the presentation logic for the initial Welcome screen.
  * It acts as an intermediary between the WelcomeView and the domain logic,
  * responding to user interactions by triggering the appropriate navigation events.
  */
 public class WelcomePresenter {
     private WelcomeView view;
 
     /**
      * Constructs a WelcomePresenter with a specific view.
      * @param view An implementation of WelcomeView to handle UI interactions.
      */
     public WelcomePresenter(WelcomeView view) {
         this.view = view;
     }
 
     /**
      * Handles the user's intent to create a new account.
      * Redirects the user to the Sign-Up screen.
      */
     public void onCreateAccount() {
         view.navigateToSignUp();
     }
 
     /**
      * Handles the user's intent to log in.
      * Redirects the user to the Login screen.
      */
     public void onLogin() {
         view.navigateToLogin();
     }
 }

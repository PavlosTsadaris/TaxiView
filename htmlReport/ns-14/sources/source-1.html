 package gr.softeng.team19.view.start.success_create;
 
 /**
  * SuccessPresenter handles the presentation logic for the account creation success screen.
  * It manages the display of confirmation messages and navigation back to the login screen.
  */
 public class SuccessPresenter {
     private SuccessView view;
 
     /**
      * Constructs a SuccessPresenter and attaches it to the specified view.
      * @param view An implementation of SuccessView to handle UI updates.
      */
     public SuccessPresenter(SuccessView view) {
         this.view = view;
     }
 
     /**
      * Initializes the view with a success message if one is provided.
      * @param message The confirmation message to be displayed on the screen.
      */
     public void onViewCreated(String message) {
         if (message != null && !message.isEmpty()) {
             view.setSuccessMessage(message);
         }
     }
 
     /**
      * Responds to the user's action to proceed to the login screen.
      */
     public void onLoginPressed() {
         view.navigateToLogin();
     }
 }

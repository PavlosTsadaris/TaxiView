 package gr.softeng.team19.view.driver.homeactivity;
 
 import android.os.Handler;
 import android.os.Looper;
 
 import gr.softeng.team19.domain.TaxiDriver;
 import gr.softeng.team19.memorydao.TaxiDriverDAOMemory;
 
 /**
  * Presenter for the Driver Home Screen.
  * <p>
  * Rationalization: This class serves as the controller for the driver's landing page.
  * It manages the display of personal stats, coordinates availability status (Online/Offline),
  * and handles conditional navigation logic. It abstracts asynchronous tasks via a
  * {@link Scheduler} interface to ensure the code remains unit-testable.
  * </p>
  */
 public class HomeDriverPresenter {
 
     /**
      * Interface to abstract task scheduling.
      * <p>
      * Logic: This abstraction allows the Presenter to schedule delayed tasks (like notifications)
      * without being strictly coupled to the Android {@link Handler}. During testing, a
      * synchronous scheduler can be injected to bypass delays.
      * </p>
      */
     public interface Scheduler {
         /**
          * Executes a task after a specified delay.
          * @param task The {@link Runnable} containing the logic to be executed.
          * @param delayMillis The delay duration in milliseconds.
          */
         void execute(Runnable task, long delayMillis);
     }
 
     private HomeDriverPresenter.Scheduler scheduler;
     private HomeDriverView view;
     private String currentUsername;
     private boolean isLocationActive = false;
 
     /**
      * Default constructor for production use.
      * <p>
      * Initializes the scheduler using the real Android Main Looper to handle
      * UI thread operations and delayed posts.
      * </p>
      * @param view The view interface implementation.
      */
     public HomeDriverPresenter(HomeDriverView view) {
         this.view = view;
         this.scheduler = (task, delay) -> new Handler(Looper.getMainLooper()).postDelayed(task, delay);
     }
 
     /**
      * Overloaded constructor for Unit Testing (Dependency Injection).
      * @param view The view stub for testing.
      * @param testScheduler A custom scheduler to execute tasks synchronously during tests.
      */
     public HomeDriverPresenter(HomeDriverView view, Scheduler testScheduler) {
         this.view = view;
         this.scheduler = testScheduler;
     }
 
     /**
      * Populates the Home Page with driver information upon loading.
      * <p>
      * Verification Logic: Fetches driver details from {@link TaxiDriverDAOMemory}.
      * If the driver is found, it updates the view with the full name and a formatted
      * rating string. If the user is missing, it reverts to guest/fallback info.
      * </p>
      * @param username The unique username of the authenticated driver.
      */
     public void onHomePageLoad(String username) {
         this.currentUsername = username;
 
         if (username == null) {
             view.setWelcomeMessage("Guest Driver");
             return;
         }
 
         TaxiDriverDAOMemory driverDAO = new TaxiDriverDAOMemory();
         TaxiDriver driver = driverDAO.find(username);
 
         if (driver != null) {
             view.setWelcomeMessage(driver.getName() + " " + driver.getSurname());
             // Formats the average rating to 1 decimal point (e.g., 4.5)
             String ratingStr = String.format("%.1f", driver.getAverageRating());
             view.setRating(ratingStr);
         } else {
             view.setWelcomeMessage(username);
             view.setRating("0.0");
         }
     }
 
     /**
      * Manages the toggling of the driver's online/offline status.
      * <p>
      * Business Logic:
      * 1. If the driver is going online, they are prompted to choose a location.
      * 2. The availability state is toggled and persisted in the {@link TaxiDriver} domain model via DAO.
      * 3. An asynchronous notification is scheduled to appear after 3 seconds,
      * provided that the driver has remained online throughout the delay.
      * </p>
      */
     public void onLocationToggleSelected() {
         if(!isLocationActive){
             view.navigateToChooseLocation();
         }
 
         isLocationActive = !isLocationActive;
 
         // Persist availability change in the Domain Model
         TaxiDriver driver = new TaxiDriverDAOMemory().find(currentUsername);
         if (driver != null) {
             driver.setAvailability(isLocationActive);
         }
 
         view.updateLocationStatus(isLocationActive);
 
         if (isLocationActive) {
             // Schedule notification via the abstracted scheduler
             scheduler.execute(() -> {
                 // Verification: Re-check if the driver is still active before showing notification
                 if (isLocationActive) {
                     view.showRequestNotification("New requests found nearby!");
                 }
             }, 3000);
         }
     }
 
     /**
      * Triggers navigation to the Driver's Profile screen.
      */
     public void onProfileSelected() {
         view.navigateToProfile(currentUsername);
     }
 
     /**
      * Triggers navigation to the Ride History screen.
      */
     public void onHistorySelected() {
         view.navigateToHistory();
     }
 
     /**
      * Handles navigation to the Request List screen.
      * <p>
      * Constraint Logic: Enforces the requirement that the driver must be
      * Online (Location Active) to access and view incoming ride requests.
      * </p>
      */
     public void onRequestsListSelected() {
         if (isLocationActive) {
             view.navigateToRequests();
         } else {
             view.showErrorMessage("You must turn on location to receive requests!");
         }
     }
 
     /**
      * Triggers navigation to the Ratings and Feedback screen.
      */
     public void onRatingsSelected() {
         view.navigateToRatings();
     }
 }
